local curTween=nil
local function toTarget(cf,spd)
    local h=hrp() if not h then return end
    spd=spd or 300
    local dist=(cf.Position-h.Position).Magnitude
    if dist<4 then return end
    if curTween then curTween:Cancel() end

-- ===== FLY (CFrame lerp — KHÔNG BodyVelocity/BodyGyro/ChangeState) =====
local Fly={On=false,Speed=100,Conn=nil,Goto=nil,
    Joy={X=0,Y=0,Up=false,Down=false},JoyUI=nil,JoyConn={}}

local function buildJoyUI(parent)
    if Fly.JoyUI or not IS_MOB then return end
    local j=make("Frame",{Size=UDim2.new(0,130,0,130),Position=UDim2.new(0,12,1,-150),
        BackgroundColor3=C.Surface,BorderSizePixel=0,Visible=false,Parent=parent})
    corner(j,8)
    local knob=make("Frame",{Size=UDim2.new(0,38,0,38),Position=UDim2.new(0.5,-19,0.5,-19),
        BackgroundColor3=C.Primary,BorderSizePixel=0,Parent=j}) corner(knob,19)
    local up=make("TextButton",{Size=UDim2.new(0,42,0,42),Position=UDim2.new(1,4,0,4),
        BackgroundColor3=C.Hi,Text="↑",Font=Enum.Font.GothamBold,TextSize=18,TextColor3=C.Text,
        AutoButtonColor=false,BorderSizePixel=0,Parent=j}) corner(up,4)
    local dn=make("TextButton",{Size=UDim2.new(0,42,0,42),Position=UDim2.new(1,4,0,52),
        BackgroundColor3=C.Hi,Text="↓",Font=Enum.Font.GothamBold,TextSize=18,TextColor3=C.Text,
        AutoButtonColor=false,BorderSizePixel=0,Parent=j}) corner(dn,4)
    local dragging=false
    local c1=j.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1 then dragging=true end
    end)
    local c2=UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseMovement) then
            local cp,sz=j.AbsolutePosition,j.AbsoluteSize
            Fly.Joy.X=math.clamp((i.Position.X-(cp.X+sz.X/2))/math.max(1,sz.X/2-8),-1,1)
            Fly.Joy.Y=math.clamp((i.Position.Y-(cp.Y+sz.Y/2))/math.max(1,sz.Y/2-8),-1,1)
            knob.Position=UDim2.new(0.5,-19+Fly.Joy.X*(sz.X/2-19),0.5,-19+Fly.Joy.Y*(sz.Y/2-19))
        end
    end)
    local c3=UIS.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1 then
            if dragging then dragging=false Fly.Joy.X=0 Fly.Joy.Y=0
                TS:Create(knob,TweenInfo.new(0.1),{Position=UDim2.new(0.5,-19,0.5,-19)}):Play() end
        end
    end)
    local function bind(b,f)
        b.InputBegan:Connect(function(i) if i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1 then Fly.Joy[f]=true end end)
        b.InputEnded:Connect(function(i) if i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1 then Fly.Joy[f]=false end end)
    end
    bind(up,"Up") bind(dn,"Down")
    Fly.JoyConn={c1,c2,c3}
    Fly.JoyUI=j
end

function Fly:Start()
    if self.Conn then self.Conn:Disconnect() end
    self.Goto=nil
    local h=hrp() if not h then return end
    if self.JoyUI then self.JoyUI.Visible=true end
    local hm=hum() if hm then hm.PlatformStand=true end
    self.Conn=RS.Heartbeat:Connect(function()
        local h2=hrp() if not h2 then return end
        -- Auto-bay tới target
        if self.Goto then
            local dir=self.Goto-h2.Position
            local dist=dir.Magnitude
            if dist<8 then self.Goto=nil notify("Bay","Đã tới nơi",2) return end
            local sp=dist<60 and self.Speed*0.6 or self.Speed
            local newPos=h2.Position+dir.Unit*math.min(dist,sp*0.1)
            h2.CFrame=h2.CFrame:Lerp(CFrame.lookAt(newPos,newPos+dir),0.5)
            h2.Velocity=Vector3.zero
            return
        end
        -- Di chuyển theo input
        local cc=Cam.CFrame
        local m=Vector3.zero
        if IS_MOB then
            m=cc.LookVector*(-self.Joy.Y)+cc.RightVector*self.Joy.X
            if self.Joy.Up then m=m+Vector3.new(0,1,0) end
            if self.Joy.Down then m=m-Vector3.new(0,1,0) end
        else
            if UIS:IsKeyDown(Enum.KeyCode.W) then m=m+cc.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.S) then m=m-cc.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.A) then m=m-cc.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.D) then m=m+cc.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.Space) then m=m+Vector3.new(0,1,0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then m=m-Vector3.new(0,1,0) end
        end
        if m.Magnitude>0.01 then
            local tgt=h2.Position+m.Unit*(self.Speed*0.1)
            h2.CFrame=h2.CFrame:Lerp(CFrame.new(tgt,tgt+cc.LookVector),0.5)
        end
        h2.Velocity=Vector3.zero  -- clear momentum mỗi frame
    end)
end

function Fly:Stop()
    if self.Conn then self.Conn:Disconnect() self.Conn=nil end
    self.Goto=nil
    if self.JoyUI then self.JoyUI.Visible=false end
    self.Joy.X=0 self.Joy.Y=0 self.Joy.Up=false self.Joy.Down=false
    local hm=hum() if hm then hm.PlatformStand=false end
end

function Fly:GotoTarget(pos)
    if not pos then return end
    self.Goto=pos
    if not self.On then self.On=true self:Start() end
end

function Fly:Toggle(v) self.On=v if v then self:Start() else self:Stop() end end

LP.CharacterAdded:Connect(function()
    task.wait(1)
    if Fly.On then Fly:Stop() task.wait(0.3) Fly:Start() end
end)

local function parseFruit(n)
    if not n then return "?","🍎" end
    local s=n:gsub("%s*%-?%s*Fruit%s*$",""):gsub("%s*%([^)]*%)%s*$","")
    s=s:match("^%s*(.-)%s*$") or s
    return s,FE[s] or "🍎"
end

local function questGui()
    local pg=LP:FindFirstChild("PlayerGui") if not pg then return nil end
    local m=pg:FindFirstChild("Main") if not m then return nil end
    return m:FindFirstChild("Quest")
end
local function questVisible() local q=questGui() if not q then return false end return q.Visible==true end
local function questTitle()
    local q=questGui() if not q then return "" end
    local ok,r=pcall(function()
        local c=q:FindFirstChild("Container") if not c then return "" end
        local t=c:FindFirstChild("QuestTitle") if not t then return "" end
        local tt=t:FindFirstChild("Title") if not tt then return "" end
        return tt.Text
    end)
    return ok and r or ""
end
local function questProgress()
    local ttl=questTitle() if ttl=="" then return nil,nil end
    local cur,max=ttl:match("(%d+)%s*/%s*(%d+)")
    if cur and max then return tonumber(cur),tonumber(max) end
    return nil,nil
end
local function questCompleted()
    local cur,max=questProgress() if not cur or not max then return false end
    if max<=0 then return false end
    return cur>=max
end

    if not name or name=="" then return end
    local c=chr() if not c then return end

    local bp=LP:FindFirstChild("Backpack") if not bp then return end

    local c=chr()

        if bp then local t=bp:FindFirstChild("Combat") or bp:FindFirstChildOfClass("Tool") if t then equip(t.Name) end end
    end
end
local lastClick=0
local function doClick()
    if tick()-lastClick<0.12 then return end  -- 8 click/sec — đủ đánh, không spam
    lastClick=tick()
    pcall(function()
        local c=chr() if not c then return end
        local t=c:FindFirstChildOfClass("Tool")
        if t then
            t:Activate()
            if t:FindFirstChild("LeftClickRemote") then
                t.LeftClickRemote:FireServer(Vector3.new(0,-1,0),1)
            end
        end
    end)
end

-- BRING MOB (set CFrame trực tiếp, không BodyVelocity, mob không bay lên trời)
-- Smart mode: chỉ kéo mob cùng tên với quest, giới hạn số lượng theo quest progress
-- mob = mob object, questName = tên mob quest cần (vd "Bandit"), maxCount = số còn cần
local broughtMobs={}  -- track các mob đang được bring (để giới hạn count)
local function bringMob(mob,questName,maxCount)
    if not mob or not mob.HRP or not mob.H or mob.H.Health<=0 then return false end
    local h=hrp() if not h then return false end
    -- Smart Bring Mob: kiểm tra quest name + count limit
    if S.BringMobSmart and questName then
        -- Chỉ bring mob đúng loại quest
        if mob.N ~= questName and mob.Full ~= questName then return false end
        -- Đếm số mob đã bring cùng loại
        local count=0
        for m,_ in pairs(broughtMobs) do
            if m and m.Parent and m:FindFirstChildOfClass("Humanoid") and m:FindFirstChildOfClass("Humanoid").Health>0 then
                local mn=baseName(m.Name)
                if mn==questName then count=count+1 end
            else
                broughtMobs[m]=nil  -- cleanup mob chết/rời đi
            end
        end
        -- Giới hạn theo quest count (nếu có)
        if maxCount and count>=maxCount then return false end
    end
    if sethiddenproperty then pcall(function() sethiddenproperty(LP,"SimulationRadius",math.huge) end) end
    pcall(function() mob.HRP:SetNetworkOwner(LP) end)
    if mob.HRP.Size.X<35 then pcall(function() mob.HRP.Size=Vector3.new(35,35,35) end) end
    pcall(function() mob.HRP.CanCollide=false end)
    local lv=h.CFrame.LookVector
    local fl=Vector3.new(lv.X,0,lv.Z)
    if fl.Magnitude<0.01 then fl=Vector3.new(0,0,-1) end
    fl=fl.Unit
    -- Vị trí mob: trước mặt player, nhưng offset nhẹ theo count để mob không chồng nhau
    local offset=#broughtMobs*1.5
    local tgt=Vector3.new(h.Position.X+fl.X*(4+offset),h.Position.Y,h.Position.Z+fl.Z*(4+offset))
    pcall(function() mob.HRP.CFrame=CFrame.new(tgt) mob.HRP.Velocity=Vector3.zero end)
    broughtMobs[mob.M]=true
    return true
end
-- Helper: clear broughtMobs table (không reassign local)
local function clearBroughtMobs()
    for k in pairs(broughtMobs) do broughtMobs[k]=nil end
end

-- Cleanup broughtMobs định kỳ (gọi từ AF loop)
local function cleanupBroughtMobs()
    for m,_ in pairs(broughtMobs) do
        if not m or not m.Parent then
            broughtMobs[m]=nil
        else
            local h=m:FindFirstChildOfClass("Humanoid")
            if not h or h.Health<=0 then broughtMobs[m]=nil end
        end
    end
end

local function attackMob(mob,questName,maxCount)
    if not mob or not mob.HRP or not mob.H or mob.H.Health<=0 then return end
    if S.Magnet then
        bringMob(mob,questName,maxCount)
    else
        local h=hrp() if h then
            local tgt=mob.HRP.CFrame*CFrame.new(0,S.FarmH,8)
            tgt=CFrame.new(tgt.X,math.max(tgt.Y,h.Position.Y-5),tgt.Z)
            toTarget(tgt,400)
        end
    end
    autoEquip() doClick()
end

-- ===== ESP MODULE SIÊU XA (khai báo trước để EVT dùng được) =====
-- Khi SuperESP on: set SimulationRadius=huge (load mob toàn server),
--                  BillboardGui MaxDistance=huge (text hiện bất kể khoảng cách)
--                  Camera FarClipPlane tăng (render xa hơn)
local ESP={Obj={}}

-- Set SimulationRadius = huge để client load object từ xa vô hạn
local function enableSuperRender()
    if sethiddenproperty then
        pcall(function() sethiddenproperty(LP,"SimulationRadius",math.huge) end)
    end
    -- Tăng camera far plane để render object xa
    pcall(function()
        if Cam then Cam.FarClipPlane=1e9 end
    end)
    -- Lighting: tắt fog để nhìn xa hơn
    pcall(function()
        if Light then Light.FogEnd=9e9 end
    end)
end

-- Loop giữ SuperESP active (mỗi 10s - server có thể reset)
local superESPLoop=nil
local function startSuperESP()
    if superESPLoop then return end
    enableSuperRender()
    superESPLoop=task.spawn(function()
        while S.SuperESP do
            enableSuperRender()
            task.wait(10)
        end
    end)
end
local function stopSuperESP()
    if superESPLoop then task.cancel(superESPLoop) superESPLoop=nil end
end

            MaxDistance=math.huge,LightInfluence=1,Parent=p}
        o.B=make("BillboardGui",bgProps)
        make("TextLabel",{Size=UDim2.new(1,0,1,0),BackgroundTransparency=1,Text=txt or inst.Name,
            Font=Enum.Font.GothamBold,TextSize=12,TextColor3=col or C.Red,
            TextStrokeTransparency=0,TextStrokeColor3=Color3.new(0,0,0),Parent=o.B})
    end
    self.Obj[inst]={o=o}
end
function ESP:Rem(inst)
    local e=self.Obj[inst] if not e then return end
    pcall(function() if e.o.H then e.o.H:Destroy() end if e.o.B then e.o.B:Destroy() end end)
    self.Obj[inst]=nil
end

local espLoop=nil
local function espStart()
    if espLoop then return end
    -- Nếu SuperESP on → set render ngay
    if S.SuperESP then startSuperESP() end
    espLoop=task.spawn(function()
        while true do
            pcall(function()
                if S.PESP then
                    for _,p in ipairs(Players:GetPlayers()) do
                        if p~=LP and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                            local ls=p:FindFirstChild("leaderstats")
                            local bn=ls and ls:FindFirstChild("Bounty")
                            local h=hrp()
                            local d=h and math.floor((p.Character.HumanoidRootPart.Position-h.Position).Magnitude) or 0
                            local txt=p.Name..(bn and bn.Value>0 and (" | "..bn.Value.." Beli") or "").." | "..d.."m"
                            ESP:Add(p.Character,C.Blue,txt)
                            local e=ESP.Obj[p.Character]
                            if e and e.o.B and e.o.B:FindFirstChildOfClass("TextLabel") then e.o.B:FindFirstChildOfClass("TextLabel").Text=txt end
                        end
                    end
                end
                if S.BESP then
                    -- Khi SuperESP: scan trực tiếp WS.Enemies (bypass MobCache distance filter)
                    if S.SuperESP then
                        local EN=WS:FindFirstChild("Enemies")
                        if EN then
                            for _,v in ipairs(EN:GetChildren()) do
                                local h=v:FindFirstChildOfClass("Humanoid")
                                if h and h.Health>0 and (v.Name:find("%[Boss%]") or v.Name:find("%[Raid Boss%]")) then
                                    ESP:Add(v,C.Red,baseName(v.Name))
                                end
                            end
                        end
                    else
                        for _,m in ipairs(MobCache) do if m.Boss then ESP:Add(m.M,C.Red,m.Full) end end
                    end
                end
                if S.MESP then
                    -- SuperESP on: không giới hạn cap, scan toàn bộ Enemies
                    if S.SuperESP then
                        local EN=WS:FindFirstChild("Enemies")
                        if EN then
                            for _,v in ipairs(EN:GetChildren()) do
                                local h=v:FindFirstChildOfClass("Humanoid")
                                local p=v:FindFirstChild("HumanoidRootPart")
                                if h and p and h.Health>0 and not (v.Name:find("%[Boss%]") or v.Name:find("%[Raid Boss%]")) then
                                    ESP:Add(v,C.Green,baseName(v.Name))
                                end
                            end
                        end
                    else
                        local n=0 local cap=S.Light and 10 or 20
                        for _,m in ipairs(MobCache) do if n<cap then ESP:Add(m.M,C.Green,m.N) n=n+1 end end
                    end
                end
                if S.CESP then
                    local CM=WS:FindFirstChild("ChestModels")
                    if CM then for _,c in ipairs(CM:GetChildren()) do ESP:Add(c,C.Yellow,"Chest: "..c.Name) end end
                end
                if S.FruitESP then
                    local h=hrp()
                    for _,t in ipairs(WS:GetChildren()) do
                        if t:IsA("Tool") and t.Name:find("Fruit") then
                            local short,emoji=parseFruit(t.Name)
                            local dist=h and t:FindFirstChild("Handle") and math.floor((t.Handle.Position-h.Position).Magnitude) or 0
                            local txt=emoji.." "..short.." | "..dist.."m"
                            ESP:Add(t,C.Green,txt)
                            local e=ESP.Obj[t]
                            if e and e.o.B then local lbl=e.o.B:FindFirstChildOfClass("TextLabel") if lbl then lbl.Text=txt end end
                        end
                    end
                end
                for i in pairs(ESP.Obj) do if i.Parent==nil then ESP:Rem(i) end end
            end)
            task.wait(S.Light and 3 or 2)
        end
    end)
end

-- ===== AUTO FARM ALL-IN-ONE (gộp Bring Mob + Magnet + Auto Click vào 1 toggle) =====
-- Khi user bật Auto Farm → tự động enable:
--   - Smart Bring Mob (kéo quái đúng loại quest)
--   - Magnet Quái (hitbox to)
--   - Auto Click (tự click liên tục theo rate)
--   - Auto Haki Buso (nếu chưa bật)
--   - Auto Stat (nếu có points dư)
-- Khi user tắt Auto Farm → tự động disable all
-- User chỉ cần 1 toggle!
local AF={Loop=nil,Fail=0,Hint=0,AutoClickLoop=nil}

function AF:Stop()
    if self.Loop then task.cancel(self.Loop) self.Loop=nil end
    if self.AutoClickLoop then task.cancel(self.AutoClickLoop) self.AutoClickLoop=nil end
    stopMove()
    clearBroughtMobs()
end

-- Sub-loop: Auto Click (chạy song song với Auto Farm)
function AF:StartClick()
    if self.AutoClickLoop then return end
    self.AutoClickLoop=task.spawn(function()
        while S.AutoFarm do
            pcall(function()
                autoEquip()
                local c=chr()
                if c then
                    local t=c:FindFirstChildOfClass("Tool")
                    if t then
                        t:Activate()
                        if t:FindFirstChild("LeftClickRemote") then
                            t.LeftClickRemote:FireServer(Vector3.new(0,-1,0),1)
                        end
                    end
                end
            end)
            task.wait(S.ClickRate)
        end
    end)
end

function AF:StartFarm()
    self:Stop()
    -- Tự động bật các chế độ phụ (Smart Bring + Magnet + Click) — không cần toggle riêng
    S.BringMobSmart=true
    S.Magnet=true
    notify("Auto Farm","Đã BẬT! Tự động hoàn toàn:\n• Quét level → nhận quest\n• Bring mob đúng loại\n• Click liên tục\n• Magnet quái\n• Auto Haki Buso (nếu chưa bật)",5)
    -- Bật Auto Click song song
    self:StartClick()
    -- Bật Auto Haki nếu user chưa bật từ Misc
    if not S.AutoHaki then
        S.AutoHaki=true
        self.HakiWasOff=true  -- nhớ để tắt lại khi Auto Farm off
        AH:Start()
    else
        self.HakiWasOff=false
    end
    self.Loop=task.spawn(function()
        while S.AutoFarm do
            pcall(function()
                if not chr() then return end
                -- Cleanup mob đã chết
                cleanupBroughtMobs()
                local lvl=getLevel()
                if tick()-self.Hint>30 then
                    if SEA==1 and lvl>=700 then notify("Auto Farm","Bạn Lv "..lvl.." — nên sang Sea 2 (tab Misc)",5) self.Hint=tick() end
                    if SEA==2 and lvl>=1500 then notify("Auto Farm","Bạn Lv "..lvl.." — nên sang Sea 3 (tab Misc)",5) self.Hint=tick() end
                end
                -- Tự quét level → chọn quest phù hợp với sea hiện tại
                local qd=getQuest(lvl,SEA) or getQuest(0,SEA)
                -- Quest đã xong → abandon để nhận quest mới cùng level
                if questVisible() and questCompleted() then
                    INV("AbandonQuest")
                    clearBroughtMobs()
                    task.wait(0.4)
                    return
                end
                if not questVisible() then
                    if self.Fail>=3 then
                        -- Fallback: farm thường không cần quest
                        local mob=findMob(qd.b)
                        if mob then attackMob(mob,qd.b,nil) else toTarget(qd.cm,300) end
                        return
                    end
                    local h=hrp() if not h then return end
                    -- Bay tới NPC nhận quest
                    if (h.Position-qd.cq.Position).Magnitude>14 then
                        toTarget(qd.cq*CFrame.new(0,0,4),320) return
                    end
                    -- Nhận quest
                    stopMove()
                    INV("SetSpawnPoint")
                    INV("StartQuest",qd.q,qd.n)
                    task.wait(0.6)
                    if not questVisible() then
                        self.Fail=self.Fail+1
                        if self.Fail==3 then notify("Auto Farm","Quest lỗi — chuyển sang farm thường",4) end
                    else
                        self.Fail=0
                        notify("Auto Farm","Đã nhận quest: "..qd.b,3)
                    end
                else
                    -- Đang có quest
                    self.Fail=0
                    local ttl=questTitle()
                    -- Quest sai loại quái → abandon để nhận lại
                    if ttl~="" and not ttl:find(qd.m,1,true) then
                        INV("AbandonQuest")
                        clearBroughtMobs()
                        task.wait(0.3)
                        return
                    end
                    -- Tính số quái còn cần = max - cur
                    local cur,max=questProgress()
                    local need = max and math.max(0, max - (cur or 0)) or nil
                    -- Tìm mob đúng loại quest
                    local mob=findMob(qd.b)
                    if mob then
                        -- Bring mob đúng loại + giới hạn theo quest count
                        -- (Click đã lo bởi AutoClickLoop song song)
                        attackMob(mob,qd.b,need)
                    else
                        -- Mob không có → bay tới spawn point của mob đó
                        toTarget(qd.cm,320)
                    end
                end
            end)
            task.wait(S.Light and 0.25 or 0.15)
        end
        -- Khi tắt Auto Farm: dừng click loop + clear mob tracking
        if self.AutoClickLoop then task.cancel(self.AutoClickLoop) self.AutoClickLoop=nil end
        stopMove()
        clearBroughtMobs()
        -- Tắt Auto Haki chỉ nếu script tự bật (HakiWasOff=true)
        if self.HakiWasOff and S.AutoHaki then
            S.AutoHaki=false
            AH:Stop()
        end
    end)
end
function AF:RunGeneric(fn)
    self:Stop()
    self.Loop=task.spawn(function()
        while true do
            pcall(function()
                if not chr() then return end
                local t=fn()
                if t then if t.m then attackMob(t.m) elseif t.p then toTarget(t.p,320) end end
            end)
            task.wait(S.Light and 0.25 or 0.15)
        end
    end)
end

local function loadState()
    if not readfile or not isfile then return nil end
    local ok, content = pcall(function()
        if not isfile(STATE_FILE) then return nil end
        return readfile(STATE_FILE)
    end)
    if not ok or not content then return nil end
    local ok2, st = pcall(function() return HS:JSONDecode(content) end)
    if not ok2 or type(st) ~= "table" then return nil end
    return st
end

local function clearState()
    if not writefile then return end
    if isfile and isfile(STATE_FILE) then
        pcall(function() delfile(STATE_FILE) end)
    end
end

-- ===== AUTO HOP SERVER TÌM TRÁI ÁC QUỶ =====
-- Khi bật: nếu server hiện tại không có fruit → hop sang server khác
-- Yêu cầu: executor hỗ trợ queue_on_teleport (để script tự chạy lại sau khi hop)
-- Sau khi hop: script tự check lại fruit → có thì báo + bay tới, không thì hop tiếp
local HopFruit={Loop=nil, HopCount=0}

local function queueAutoExec()
    -- Queue script để tự chạy lại khi teleport (cần Auto Execute trong executor)
    -- User cần set getgenv().TNT_SCRIPT_URL = "URL_script" trước khi load TNThud
    -- VD: getgenv().TNT_SCRIPT_URL = "https://pastebin.com/raw/xxx" rồi loadstring(...)
    if queue_on_teleport then
        pcall(function()
            local scriptUrl = getgenv() and getgenv().TNT_SCRIPT_URL
            if scriptUrl and scriptUrl ~= "" then
                queue_on_teleport([[loadstring(game:HttpGet("]]..scriptUrl..[["))()]])
            else
                -- Không có URL: queue_on_teleport không làm gì (executor tự auto-exec script nếu có Auto Exec)
                -- User chỉ cần copy script vào Auto Exec folder của executor
                queue_on_teleport("")
            end
        end)
    end
end

local function hopToNextServer()
    local req
    pcall(function() req = syn and syn.request or request or http_request end)
    if not req then
        notify("Hop Fruit","Executor không hỗ trợ HTTP — không thể hop",4)
        return false
    end
    -- Lưu state trước khi hop (nếu AutoRestore on)
    if S.AutoRestore then saveState() end
    -- Lưu flag hop mode qua getgenv để script sau khi teleport biết cần hunt fruit
    if getgenv then
        getgenv().TNT_HOP_MODE="fruit"
        getgenv().TNT_HOP_COUNT=(getgenv().TNT_HOP_COUNT or 0)+1
    end
    -- Queue script tự chạy lại
    queueAutoExec()
    -- Tìm server khác
    local hopped=false
    pcall(function()
        local url="https://games.roblox.com/v1/games/"..game.PlaceId.."/servers/Public?sortOrder=Asc&limit=100"
        local data=HS:JSONDecode(req({Url=url, Method="GET"}).Body)
        for _,sv in ipairs(data.data) do
            if sv.playing<sv.maxPlayers and sv.id~=game.JobId then
                notify("Hop Fruit","Đang hop tới server khác (lần "..(getgenv().TNT_HOP_COUNT or 0)..")...",3)
                TPS:TeleportToPlaceInstance(game.PlaceId, sv.id, LP)
                hopped=true
                return
            end
        end
    end)
    return hopped
end

function HopFruit:Start()
    if self.Loop then return end
    notify("Auto Hop Fruit","BẬT! Sẽ tự tìm trái ác quỷ qua nhiều server\n(yêu cầu bật Auto Execute trong executor)",5)
    self.Loop=task.spawn(function()
        while S.AutoHopFruit do
            pcall(function()
                local f=findFruit()
                if f and f:FindFirstChild("Handle") then
                    -- Có fruit → báo + bay tới
                    local short,emoji=parseFruit(f.Name)
                    notify("TÌM THẤY!"," "..emoji.." "..short.." ở server này! Đang bay tới...",8)
                    if not Fly.On then
                        Fly:GotoTarget(f.Handle.Position)
                    end
                    -- Bật Fruit ESP nếu chưa
                    if not S.FruitESP then
                        S.FruitESP=true
                        espStart()
                    end
                    -- Clear hop mode
                    if getgenv then getgenv().TNT_HOP_MODE=nil end
                    -- Dừng hop (nhưng giữ Auto Hop toggle để user thấy trạng thái)
                    S.AutoHopFruit=false
                    return
                end
                -- Không có fruit → hop sau 3s
                task.wait(3)
                if S.AutoHopFruit then
                    hopToNextServer()
                end
            end)
            task.wait(2)
        end
    end)
end
function HopFruit:Stop()
    if self.Loop then task.cancel(self.Loop) self.Loop=nil end
    if getgenv then getgenv().TNT_HOP_MODE=nil end
end

function Misc:FPS()
    pcall(function()
        local t=WS.Terrain
        t.WaterWaveSize=0 t.WaterWaveSpeed=0 t.WaterReflectance=0 t.WaterTransparency=0
        Light.GlobalShadows=false Light.FogEnd=9e9 Light.Brightness=0
        settings().Rendering.QualityLevel=Enum.QualityLevel.Level01
        for _,o in ipairs(WS:GetDescendants()) do
            if o:IsA("BasePart") then o.Material=Enum.Material.Plastic o.Reflectance=0
            elseif o:IsA("Decal") or o:IsA("Texture") then o.Transparency=1
            elseif o:IsA("ParticleEmitter") or o:IsA("Trail") then o.Lifetime=NumberRange.new(0)
            elseif o:IsA("Fire") or o:IsA("SpotLight") or o:IsA("Smoke") or o:IsA("Sparkles") or o:IsA("Explosion") then o.Enabled=false
            elseif o:IsA("MeshPart") then o.Material=Enum.Material.Plastic o.Reflectance=0 end
        end
        for _,e in ipairs(Light:GetChildren()) do
            if e:IsA("BlurEffect") or e:IsA("SunRaysEffect") or e:IsA("ColorCorrectionEffect") or e:IsA("BloomEffect") or e:IsA("DepthOfFieldEffect") then e.Enabled=false end
        end
    end)
    notify("FPS Boost","Đã áp dụng!",3)
end

-- ===== RESUME HOP MODE (khi script tự chạy lại sau khi teleport) =====
-- Nếu trước khi hop, user đã bật AutoHopFruit → script load xong sẽ tiếp tục hunt
task.spawn(function()
    task.wait(3)  -- đợi script load xong + character spawn
    if getgenv and getgenv().TNT_HOP_MODE == "fruit" then
        local count = getgenv().TNT_HOP_COUNT or 0
        if count > 20 then
            -- Giới hạn 20 lần hop để tránh infinite loop
            getgenv().TNT_HOP_MODE=nil
            getgenv().TNT_HOP_COUNT=0
            notify("Hop Fruit","Đã thử 20 server, không tìm thấy trái ác quỷ — dừng lại",6)
            return
        end
        notify("Hop Fruit","Script tự chạy lại! Đang kiểm tra server mới (lần "..count..")...",5)
        -- Đợi character + workspace load
        task.wait(2)
        local f = findFruit()
        if f and f:FindFirstChild("Handle") then
            -- TÌM THẤY!
            local short, emoji = parseFruit(f.Name)
            notify("TÌM THẤY TRÁI ÁC QUỶ!"," "..emoji.." "..short.." ở server này!",8)
            if not Fly.On then
                Fly:GotoTarget(f.Handle.Position)
            end
            if not S.FruitESP then
                S.FruitESP = true
                espStart()
            end
            getgenv().TNT_HOP_MODE = nil
            getgenv().TNT_HOP_COUNT = 0
        else
            -- Không có fruit → hop tiếp
            notify("Hop Fruit","Server này không có trái — đang hop tiếp...",3)
            task.wait(2)
            hopToNextServer()
        end
    end
end)

            if st.AutoMastery then S.AutoMastery=true AF:StartMastery() end
            if st.AutoMat then S.AutoMat=true AF:StartMat() end
            if st.AutoBone then S.AutoBone=true AF:StartBone() end
            if st.AutoChest then S.AutoChest=true AF:StartChest() end
            if st.AutoElite then S.AutoElite=true AF:StartElite() end
            if st.AutoBoss then S.AutoBoss=true S.AutoBossAll=true BF:Start("",true) end

            if st.AutoAwaken then S.AutoAwaken=true AW:Start() end
